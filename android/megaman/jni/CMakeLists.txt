cmake_minimum_required(VERSION 3.18)
project(megarun CXX C)
set(CMAKE_CXX_STANDARD 17)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
# SDL is fetched once into app/jni/SDL (see fetch_sdl.sh) and shared by both modules
set(SDLDIR ${CMAKE_CURRENT_SOURCE_DIR}/SDL)
add_subdirectory(${SDLDIR} ${CMAKE_BINARY_DIR}/sdl)
set(CORE ${CMAKE_CURRENT_SOURCE_DIR}/../../../core)
add_library(main SHARED ${CORE}/main.cpp ${CORE}/cpu.cpp ${CORE}/machine.cpp ${CORE}/hw.cpp ${CORE}/gx.cpp ${CORE}/gl.cpp ${CORE}/ax.cpp ${CORE}/jit.cpp ${CORE}/jit_rt.cpp ${CORE}/wii.cpp ${CORE}/wiibt.cpp)
target_compile_options(main PRIVATE -O2 -fno-stack-protector -Wno-unused -Wno-invalid-offsetof)
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/shim)
file(CREATE_LINK ${SDLDIR}/include ${CMAKE_BINARY_DIR}/shim/SDL2 SYMBOLIC)
target_include_directories(main PRIVATE ${CMAKE_BINARY_DIR}/shim)
target_link_libraries(main SDL2 GLESv2 log)
